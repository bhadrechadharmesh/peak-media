    72→/* ------------------------------ helpers ------------------------------ */
    73→
    74→function isStepValid(step: number, f: FormState): boolean {
    75→  if (step === 0) {
    76→    return (
    77→      f.name.trim().length > 1 &&
    78→      EMAIL_RE.test(f.email) &&
    79→      f.company.trim().length > 0 &&
    80→      f.role.trim().length > 0
    81→    );
    82→  }
    83→  if (step === 1) {
    84→    return f.services.length > 0 && f.budget.length > 0;
    85→  }
    86→  // step 2
    87→  return !!f.date && f.timeSlot.length > 0;
    88→}

   326→                                placeholder="Jordan Avery"

   331→                              label="Work email"
   332→                              htmlFor="cf-email"

   343→                                  set("email", e.target.value)

   360→                                value={form.company}
   361→                                onChange={(e) =>
   362→                                  set("company", e.target.value)
   363→                                }
   364→                                placeholder="Northwind Inc."

   366→                              />
   367→                            </Field>

   369→                              <Input
   370→                                id="cf-role"
   371→                                value={form.role}

   373→                                  set("role", e.target.value)

   375→                                placeholder="Head of Growth"
   376→                                autoComplete="organization-title"
   377→                              />
