    73→    73→
    74→    74→function isStepValid(step: number, f: FormState): boolean {
    75→    75→  if (step === 0) {
    76→    76→    return (
    77→    77→      f.name.trim().length > 1 &&
    78→    78→      EMAIL_RE.test(f.email) &&
    79→    79→      f.company.trim().length > 0 &&
    80→    80→      f.role.trim().length > 0
    81→    81→    );
    82→    82→  }
    83→    83→  if (step === 1) {
    84→    84→    return f.services.length > 0 && f.budget.length > 0;
    85→    85→  }
    86→    86→  // step 2
    87→    87→  return !!f.date && f.timeSlot.length > 0;
    88→    88→}
